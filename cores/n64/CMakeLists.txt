cmake_minimum_required(VERSION 3.20)
project(n64_plugin LANGUAGES CXX)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(VELOCE_CORES_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin/cores)
endif()
add_library(n64_plugin SHARED src/plugin.cpp src/machine.cpp src/vr4300.cpp)
option(VELOCE_N64_NATIVE_OPTIMIZE "Build only the N64 plugin for this host CPU" OFF)
if(VELOCE_N64_NATIVE_OPTIMIZE)
    if(CMAKE_CROSSCOMPILING OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "N64 native optimization requires a local GCC or Clang build")
    endif()
    target_compile_options(n64_plugin PRIVATE -march=native)
endif()
set(VELOCE_N64_PGO "OFF" CACHE STRING "N64 profile-guided build: OFF, GENERATE, or USE")
set_property(CACHE VELOCE_N64_PGO PROPERTY STRINGS OFF GENERATE USE)
if(NOT VELOCE_N64_PGO STREQUAL "OFF")
    if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR CMAKE_CROSSCOMPILING)
        message(FATAL_ERROR "N64 PGO currently requires a local GCC build")
    endif()
    if(NOT IS_ABSOLUTE "${VELOCE_N64_PGO_DATA}")
        message(FATAL_ERROR "Set VELOCE_N64_PGO_DATA to an absolute, writable directory")
    endif()
    if(VELOCE_N64_PGO STREQUAL "GENERATE")
        target_compile_options(n64_plugin PRIVATE "-fprofile-generate=${VELOCE_N64_PGO_DATA}")
        target_link_options(n64_plugin PRIVATE "-fprofile-generate=${VELOCE_N64_PGO_DATA}")
    elseif(VELOCE_N64_PGO STREQUAL "USE")
        target_compile_options(n64_plugin PRIVATE "-fprofile-use=${VELOCE_N64_PGO_DATA}" -fprofile-correction -Werror=missing-profile)
        target_link_options(n64_plugin PRIVATE "-fprofile-use=${VELOCE_N64_PGO_DATA}")
    else()
        message(FATAL_ERROR "VELOCE_N64_PGO must be OFF, GENERATE, or USE")
    endif()
endif()
target_compile_features(n64_plugin PRIVATE cxx_std_17)
target_include_directories(n64_plugin PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../include)
if(TARGET imgui)
    target_include_directories(n64_plugin PRIVATE ${imgui_SOURCE_DIR})
    target_link_libraries(n64_plugin PRIVATE imgui)
    target_compile_definitions(n64_plugin PRIVATE VELOCE_N64_IMGUI=1)
endif()
set_target_properties(n64_plugin PROPERTIES PREFIX "" OUTPUT_NAME "n64"
    LIBRARY_OUTPUT_DIRECTORY ${VELOCE_CORES_OUTPUT_DIRECTORY}
    RUNTIME_OUTPUT_DIRECTORY ${VELOCE_CORES_OUTPUT_DIRECTORY})
foreach(config IN ITEMS Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER ${config} upper)
    set_target_properties(n64_plugin PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY_${upper} ${VELOCE_CORES_OUTPUT_DIRECTORY}
        RUNTIME_OUTPUT_DIRECTORY_${upper} ${VELOCE_CORES_OUTPUT_DIRECTORY})
endforeach()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(n64_plugin PRIVATE -Wall -Wextra -Wpedantic)
elseif(MSVC)
    target_compile_options(n64_plugin PRIVATE /W4)
endif()
